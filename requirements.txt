# Yev Capital | LoanLogic - runtime dependencies
#
# Lower bounds are the oldest releases this app is tested against; the upper
# caps keep a future major release from silently changing behaviour on a
# redeploy. Exact pins are deliberately avoided: an old pinned build (numpy in
# particular) fails to compile on newer Python versions, which is a common way
# for a hosted deployment to break on an environment rebuild.
#
# Verified against Streamlit 1.60.0 and 1.62.0.

streamlit>=1.30,<2.0        # UI framework; version shims cover 1.30 -> 1.62
pandas>=2.0,<4.0            # schedule dataframes (pulls in numpy + dateutil)
plotly>=5.18,<8.0           # interactive charts
openpyxl>=3.1,<4.0          # .xlsx export
python-docx>=1.0,<2.0       # .docx export
reportlab>=4.0,<6.0         # .pdf export
python-dateutil>=2.8,<3.0   # relativedelta - month-end-safe date arithmetic

# NOTE: `scipy` used to be listed here but is never imported anywhere in the
# codebase. It is one of the largest wheels in the tree, so carrying it made
# every environment rebuild slower (and more failure-prone) for no benefit.
# `numpy` is likewise never imported directly - pandas requires it, so it
# still arrives transitively and numpy scalars keep formatting correctly.
